/* ===== CHAT ROOM ===== */ let chatMessages = []; function openChatRoom() { const room = document.getElementById("liveBox"); if (!room) return; room.style.display = "block"; let chatBox = document.getElementById("chatRoomBox"); if (!chatBox) { chatBox = document.createElement("div"); chatBox.id = "chatRoomBox"; chatBox.innerHTML = `

💬 Chat Room

`; room.appendChild(chatBox); document.getElementById("chatInput").addEventListener( "keydown", function(e) { if (e.key === "Enter") { sendChatMessage(); } } ); } } function sendChatMessage() { const input = document.getElementById("chatInput"); const messages = document.getElementById("chatMessages"); if (!input || !messages) return; const text = input.value.trim(); if (!text) return; const message = { text: text, time: new Date().toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }) }; chatMessages.push(message); const item = document.createElement("div"); item.style.cssText = ` background:#191b2c; padding:10px 12px; margin-bottom:8px; border-radius:15px; word-break:break-word; `; item.innerHTML = `
👤 You
${escapeChatText(message.text)}
${message.time} `; messages.appendChild(item); messages.scrollTop = messages.scrollHeight; input.value = ""; input.focus(); } function clearChatMessages() { chatMessages = []; const messages = document.getElementById("chatMessages"); if (messages) { messages.innerHTML = ""; } } function escapeChatText(text) { const div = document.createElement("div"); div.textContent = text; return div.innerHTML; } /* Chat button کو Chat Room کھولنے سے جوڑیں */ document.addEventListener("DOMContentLoaded", function() { const chatButtons = document.querySelectorAll( '[data-page="chat"], .chat, #chatBtn' ); chatButtons.forEach(function(button) { button.addEventListener("click", function() { openChatRoom(); }); }); });
